### 步骤2: 查看业务层
```python
# csrcqsf/biz/railwayparts/parts.py:556
@classmethod
async def rebuild_pic_index(cls):
    """
    重建所有配件的图片索引
    
    核心算法流程：
    1. 标记所有图片为未索引状态（数据库操作）
    2. 获取所有图片记录
    3. 使用DINOv2提取特征并重建Milvus向量索引
    4. 标记所有图片为已索引状态
    5. 提交数据库事务
    
    技术栈：
    - DINOv2-Large: 提取1024维特征向量
    - Milvus: 向量数据库
    - PostgreSQL: 存储索引状态
    """
    async with SessionLocal() as session:
        # 步骤1: 标记所有图片为未索引状态
        await SgoItemImages.make_image_unindexed(session=session)
        
        # 步骤2: 获取所有图片记录
        images = SgoItemImages.get_all_image(session=session)
        
        # 步骤3: 重建Milvus向量索引（核心算法）
        await MilvusManager.rebuild_collection([image async for image in images])
        
        # 步骤4: 标记所有图片为已索引状态
        await SgoItemImages.make_image_indexed(session=session)
        
        # 步骤5: 提交事务
        await session.commit()
```

### 步骤3: 查看数据层
```python
# csrcqsf/models/sgo.py:135
class SgoItemImages(Base):
    """
    配件图片信息表
    
    字段说明：
    - is_indexed: 索引状态（0=未索引, 1=已索引, 2=初始状态）
    """
    
    @classmethod
    async def make_image_unindexed(cls,session:AsyncSession):
        """标记所有图片为未索引状态"""
        stmt = update(cls).values(is_indexed=0)
        await session.execute(stmt)
        await session.flush()
```
